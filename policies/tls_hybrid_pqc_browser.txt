# TLS 1.3: hybrid classical + ML-KEM first, then classical groups.
# Firefox 132+ and Chrome 131+ offer X25519MLKEM768. The server picks the
# first group in this list that the client also offers, so a browser that
# supports the hybrid negotiates x25519/ML-KEM-768.
# Pure ML-KEM is not in this file: browsers do not offer it and the handshake
# fails with SSL_ERROR_NO_CYPHER_OVERLAP. Use pqc_basic.txt for Botan clients.

allow_tls13 = true
allow_tls12 = true
ciphers = ChaCha20Poly1305 AES-256/GCM AES-128/GCM
key_exchange_groups = x25519/ML-KEM-768 x25519 secp256r1 secp384r1
require_cert_revocation_info = false
